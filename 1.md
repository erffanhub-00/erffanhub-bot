var { categories, tools } = tools_default;
function buildMainMenu() {
  const rows = categories.map((category) => [
    { text: `${category.emoji} ${category.name}`.trim(), callback_data: `cat:${category.id}` }
  ]);
  rows.push([
    { text: "👤 About", callback_data: "about" },
    { text: "📖 Help", callback_data: "help" }
  ]);
  rows.push([{ text: "🔗 Links", callback_data: "links" }]);
  return { inline_keyboard: rows };
}
function buildCategoryMenu(categoryId) {
  const categoryTools = tools.filter((tool) => tool.category === categoryId);
  const rows = categoryTools.map((tool) => [
    { text: `🔗 ${tool.name}`, url: tool.url }
  ]);
  rows.push([{ text: "⬅️ Back", callback_data: "back:main" }]);
  return { inline_keyboard: rows };
}
function buildAboutMenu(links) {
  const linkRows = links.map((link) => [
    { text: `🔗 ${link.label}`, url: link.url }
  ]);
  linkRows.push([{ text: "⬅️ Back", callback_data: "back:main" }]);
  return { inline_keyboard: linkRows };
}
function getMainMenuText() {
  return "📦 *ErffanHub Toolbox*\n\nSelect a category:";
}
function getCategoryText(categoryId) {
  const category = categories.find((c) => c.id === categoryId);
  if (!category) return "❌ Category not found.";
  const categoryTools = tools.filter((tool) => tool.category === categoryId);
  if (categoryTools.length === 0) {
    return `📂 *${category.name}*\n\n⚠️ No tools in this category yet.`;
  }
  return `📂 *${category.name}*\n\nSelect a tool:`;
}
function getLinksText() {
  return "🔗 *Links*\n\nFind me on:";
}

async function callTelegram(method, payload, env) {
  const url = `https://api.telegram.org/bot${env.BOT_TOKEN}/${method}`;
  let response;
  try {
    response = await fetch(url, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(payload)
    });
  } catch (err) {
    console.error(`Network error on ${method}: ${err.message}`);
    return null;
  }
  if (!response.ok) {
    console.error(`Telegram HTTP error on ${method}: ${response.status}`);
    return null;
  }
  let body;
  try {
    body = await response.json();
  } catch {
    console.error(`Invalid JSON from Telegram on ${method}`);
    return null;
  }
  if (!body.ok) {
    console.error(`Telegram API rejected ${method}: ${body.description || "unknown error"}`);
    return null;
  }
  return body.result;
}
function isOwner(userId, env) {
  return String(userId) === String(env.OWNER_TELEGRAM_ID);
}
async function handleUpdate(update, env) {
  if (update.message?.text) {
    await handleMessage(update.message, env);
    return;
  }
  if (update.callback_query) {
    await handleCallbackQuery(update.callback_query, env);
  }
}
async function handleMessage(message, env) {
  const chatId = message.chat.id;
  const chatType = message.chat.type;
  const userId = message.from.id;
  const text = message.text.trim();
  if (!isOwner(userId, env)) {
    if (chatType === "private") {
      await callTelegram("sendMessage", { chat_id: chatId, text: "⛔ Access denied." }, env);
    }
    return;
  }
  if (text === "/start") {
    await callTelegram("sendMessage", {
      chat_id: chatId,
      text: getMainMenuText(),
      parse_mode: "Markdown",
      reply_markup: buildMainMenu()
    }, env);
    return;
  }
  if (text === "/help") {
    await callTelegram("sendMessage", {
      chat_id: chatId,
      text: getHelpText(),
      parse_mode: "Markdown",
      reply_markup: { inline_keyboard: [[{ text: "⬅️ Back", callback_data: "back:main" }]] }
    }, env);
    return;
  }
  if (text === "/about") {
    await callTelegram("sendMessage", {
      chat_id: chatId,
      text: getAboutText(),
      parse_mode: "Markdown",
      reply_markup: buildAboutMenu(owner.links)
    }, env);
    return;
  }
  if (text === "/links") {
    await callTelegram("sendMessage", {
      chat_id: chatId,
      text: getLinksText(),
      parse_mode: "Markdown",
      reply_markup: buildAboutMenu(owner.links)
    }, env);
    return;
  }
  await callTelegram("sendMessage", { chat_id: chatId, text: "Invalid command. Use /start." }, env);
}
async function handleCallbackQuery(query, env) {
  const chatId = query.message.chat.id;
  const userId = query.from.id;
  const messageId = query.message.message_id;
  const data = query.data || "";
  if (!isOwner(userId, env)) {
    if (query.message.chat.type === "private") {
      await callTelegram("answerCallbackQuery", {
        callback_query_id: query.id,
        text: "⛔ Access denied",
        show_alert: true
      }, env);
    } else {
      await callTelegram("answerCallbackQuery", { callback_query_id: query.id }, env);
    }
    return;
  }
  await callTelegram("answerCallbackQuery", { callback_query_id: query.id }, env);
  if (data.startsWith("cat:")) {
    const categoryId = data.slice(4);
    const categoryMenu = buildCategoryMenu(categoryId);
    await callTelegram("editMessageText", {
      chat_id: chatId,
      message_id: messageId,
      text: getCategoryText(categoryId),
      parse_mode: "Markdown",
      reply_markup: categoryMenu
    }, env);
    return;
  }
  if (data === "about") {
    await callTelegram("editMessageText", {
      chat_id: chatId,
      message_id: messageId,
      text: getAboutText(),
      parse_mode: "Markdown",
      reply_markup: buildAboutMenu(owner.links)
    }, env);
    return;
  }
  if (data === "help") {
    await callTelegram("editMessageText", {
      chat_id: chatId,
      message_id: messageId,
      text: getHelpText(),
      parse_mode: "Markdown",
      reply_markup: { inline_keyboard: [[{ text: "⬅️ Back", callback_data: "back:main" }]] }
    }, env);
    return;
  }
  if (data === "links") {
    await callTelegram("editMessageText", {
      chat_id: chatId,
      message_id: messageId,
      text: getLinksText(),
      parse_mode: "Markdown",
      reply_markup: buildAboutMenu(owner.links)
    }, env);
    return;
  }
  if (data === "back:main") {
    await callTelegram("editMessageText", {
      chat_id: chatId,
      message_id: messageId,
      text: getMainMenuText(),
      parse_mode: "Markdown",
      reply_markup: buildMainMenu()
    }, env);
  }
}

export default {
  async fetch(request, env, ctx) {
    const url = new URL(request.url);

    if (url.pathname === "/setup-commands") {
      const res = await fetch(
        `https://api.telegram.org/bot${env.BOT_TOKEN}/setMyCommands`,
        {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            scope: { type: "default" },
            commands: [
              { command: "start", description: "Open the main menu" },
              { command: "help", description: "Show help" },
              { command: "about", description: "About the owner" },
              { command: "links", description: "Social links" }
            ]
          })
        }
      );
      const text = await res.text();
      return new Response(text, { status: 200 });
    }

    if (url.pathname === "/get-commands") {
      const res = await fetch(
        `https://api.telegram.org/bot${env.BOT_TOKEN}/getMyCommands`,
        {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ scope: { type: "default" } })
        }
      );
      const text = await res.text();
      return new Response(text, { status: 200 });
    }

    if (request.method === "POST" && url.pathname === "/webhook") {
      if (env.WEBHOOK_SECRET) {
        const incomingSecret = request.headers.get("X-Telegram-Bot-Api-Secret-Token");
        if (incomingSecret !== env.WEBHOOK_SECRET) {
          return new Response("Forbidden", { status: 403 });
        }
      }
      try {
        const update = await request.json();
        ctx.waitUntil(handleUpdate(update, env));
      } catch {}
      return new Response("OK", { status: 200 });
    }

    return new Response("ErffanHub Toolbox", { status: 200 });
  }
};
